# --- MA'LUMOTLARNI TAYYORLASH ---
# Bu blok faqat bir marta, hisobot yaratilayotganda ishlaydi.
# Sizning kodingizdan nusxa olindi va biroz to'g'irlandi.

kobo_data <- readRDS("kobo_data.rds")
child_table_name <- "hh_member"

# Jadvallarni ajratish
household_raw <- kobo_data
household_raw[[child_table_name]] <- NULL 
temp_for_unnest <- kobo_data[c("_uuid", child_table_name)]
hh_members_raw <- temp_for_unnest %>% unnest(all_of(child_table_name), names_sep = "/")

# Nomlarni o'zgartirish
dict_household <- read_excel("nomlar_lugati_household.xlsx") %>% filter(!is.na(yangi_nom))
rename_vec_hh <- setNames(dict_household$eski_nom, dict_household$yangi_nom)
household_clean <- household_raw %>% rename(any_of(rename_vec_hh))

dict_hh_members <- read_excel("nomlar_lugati_hh_members.xlsx") %>% filter(!is.na(yangi_nom))
rename_vec_members <- setNames(dict_hh_members$eski_nom, dict_hh_members$yangi_nom)
hh_members_clean <- hh_members_raw %>% rename(any_of(rename_vec_members))

# Yangi ustunlarni yaratish (recode)
household_clean <- household_clean %>%
  mutate(
    туман = case_when(
      district == 1  ~ "Конимех тумани", district == 2  ~ "Қизилтепа тумани",
      district == 3  ~ "Навбаҳор тумани", district == 4  ~ "Кармана тумани",
      district == 5  ~ "Нурота тумани", district == 6  ~ "Томди тумани",
      district == 7  ~ "Учқудуқ тумани", district == 8  ~ "Хатирчи тумани",
      district == 9  ~ "Навоий шаҳри", district == 10  ~ "Зарафшон шаҳри",
      district == 11  ~ "Ғозғон шаҳри", .default = as.character(district)
    ),
    аҳоли_пункти = case_when(punkt == 1 ~ "Шаҳар", punkt == 2 ~ "Қишлоқ"),
    сўзлашув_тили = case_when(
      family_language == '_' ~ "Ўзбек", family_language == '__1' ~ "Тожик",
      family_language == '__2' ~ "Қозоқ", family_language == '__3' ~ "Қирғиз",
      family_language == '__4' ~ "Қорақалпоқ", family_language == '__5' ~ "Рус",
      family_language == '__6' ~ "Бошқа"
    )
  )

# Vaqtni hisoblash uchun ustunlarni yaratish
household_final <- household_clean %>%
  mutate(
    sana = as.Date(`_submission_time`),
    start_time = ymd_hms(start),
    end_time = ymd_hms(end),
    duration_minutes = as.numeric(difftime(end_time, start_time, units = "mins"))
  ) %>%
  filter(duration_minutes >= 1 & duration_minutes <= 180)

plotOutput("daily_submissions_plot")
# 2. Grafikni chizadigan va filtr o'zgarganda uni yangilaydigan kod (server mantig'i)
renderPlot({
  
  # shared_household_data'dan foydalanganimiz uchun, bu kod bloki
  # har safar filtr o'zgarganda avtomatik qayta ishlaydi.
  
  # Filtrlangan ma'lumotlar asosida kunlik sonlarni hisoblaymiz
  daily_summary <- shared_household_data$data(withSelection = TRUE) %>%
    filter(selected_ | is.null(filtered) || !any(filtered)) %>% # Filtrlangan qatorlarni olamiz
    count(sana, name = "soni")

  # Grafikni chizamiz
  ggplot(daily_summary, aes(x = sana, y = soni)) +
    geom_col(fill = "#28B463") +
    geom_text(aes(label = soni), vjust = -0.5) +
    scale_x_date(date_breaks = "1 day", date_labels = "%b %d") +
    labs(
      title = "Kunlik topshirilgan so'rovnomalar soni",
      x = "Sana",
      y = "So'rovnomalar soni"
    ) +
    theme_minimal(base_size = 14) +
    theme(axis.text.x = element_text(angle = 45, hjust = 1))
})
